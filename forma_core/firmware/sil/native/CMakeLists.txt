cmake_minimum_required(VERSION 3.20)
project(forma_firmware_sil LANGUAGES CXX)
add_library(forma_firmware_sil SHARED controller.cpp bridge.cpp)
target_compile_features(forma_firmware_sil PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(forma_firmware_sil PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(forma_firmware_sil PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
